#!/usr/bin/env python3
"""Independent CSV-only audit for X15; imports no CoPE code."""

from __future__ import annotations

import csv
import hashlib
import json
from collections import Counter, defaultdict
from pathlib import Path


ROOT = Path("research/generic_typed_interpreter_2026-08-03")
RUN = ROOT / "run_v1"
ARMS = {"cope_exact", "cope_generic", "compact_tx", "fsr_pc"}


def read_csv(path: Path) -> list[dict[str, str]]:
    with path.open(newline="", encoding="utf-8") as stream:
        return list(csv.DictReader(stream))


def truth(value: str) -> bool:
    if value not in {"True", "False"}:
        raise ValueError(f"not a CSV boolean: {value!r}")
    return value == "True"


def add(checks: list[dict[str, str]], name: str, passed: bool, observed: object, expected: object) -> None:
    checks.append({
        "check": name,
        "pass": str(bool(passed)),
        "observed": json.dumps(observed, sort_keys=True, separators=(",", ":")),
        "expected": json.dumps(expected, sort_keys=True, separators=(",", ":")),
    })


def status_map(path: Path) -> dict[str, str]:
    result = {}
    for line in path.read_text(encoding="utf-8").splitlines():
        key, value = line.split("=", 1)
        result[key] = value
    return result


def main() -> int:
    manifest = read_csv(ROOT / "01_FAULT_MANIFEST.csv")
    source = read_csv(RUN / "02_SOURCE_AUDIT.csv")
    clean = read_csv(RUN / "03_CLEAN_CONTROLS.csv")
    faults = read_csv(RUN / "04_FAULT_RESULTS.csv")
    aggregate = read_csv(RUN / "05_AGGREGATE.csv")
    status = status_map(RUN / "06_STATUS.txt")
    checks: list[dict[str, str]] = []

    fault_ids = [row["fault_id"] for row in manifest]
    axis_counts = Counter(row["threat_axis"] for row in manifest)
    add(checks, "manifest_22_unique", len(fault_ids) == 22 and len(set(fault_ids)) == 22, [len(fault_ids), len(set(fault_ids))], [22, 22])
    add(checks, "manifest_axis_counts", axis_counts == {"commission_or_structure": 14, "omission": 8}, dict(axis_counts), {"commission_or_structure": 14, "omission": 8})
    source_ok = len(source) == 1 and source[0]["pass"] == "True" and source[0]["forbidden_hits"] == "" and source[0]["forbidden_symbols"] == "expected_patch;derive_post_state;validate_and_compile"
    add(checks, "generic_source_forbidden_symbol_audit", source_ok, source, "one passing row with zero forbidden hits")

    clean_groups: dict[str, list[dict[str, str]]] = defaultdict(list)
    for row in clean:
        clean_groups[row["case_id"]].append(row)
    clean_shape = len(clean) == 48 and len(clean_groups) == 12 and all(len(rows) == 4 and {row["arm"] for row in rows} == ARMS for rows in clean_groups.values())
    add(checks, "clean_12_four_arm_groups", clean_shape, [len(clean), len(clean_groups)], [48, 12])
    clean_ok = all(
        truth(row["pass"]) and truth(row["parser_accepted"])
        and truth(row["native_materializer_accepted"])
        and truth(row["common_validator_accepted"])
        and row["common_validator_calls"] == "1"
        and not truth(row["proposal_changed"])
        and not truth(row["candidate_differs_oracle"])
        and truth(row["caller_state_unchanged"])
        for row in clean
    )
    add(checks, "clean_48_of_48_exact", clean_ok, sum(truth(row["pass"]) for row in clean), 48)

    fault_groups: dict[str, list[dict[str, str]]] = defaultdict(list)
    for row in faults:
        fault_groups[row["fault_id"]].append(row)
    fault_shape = len(faults) == 88 and set(fault_groups) == set(fault_ids) and all(len(rows) == 4 and {row["arm"] for row in rows} == ARMS for rows in fault_groups.values())
    add(checks, "fault_22_four_arm_groups", fault_shape, [len(faults), len(fault_groups)], [88, 22])
    add(checks, "fault_proposals_all_changed", all(truth(row["proposal_changed"]) and row["proposal_sha256"] != row["canonical_proposal_sha256"] for row in faults), sum(truth(row["proposal_changed"]) for row in faults), 88)
    add(checks, "faults_all_rejected_end_to_end", all(truth(row["full_pipeline_rejected"]) and not truth(row["common_validator_accepted"]) for row in faults), sum(truth(row["full_pipeline_rejected"]) for row in faults), 88)
    add(checks, "zero_end_to_end_fail_open", not any(truth(row["end_to_end_fail_open"]) for row in faults), sum(truth(row["end_to_end_fail_open"]) for row in faults), 0)
    add(checks, "zero_crash", not any(truth(row["uncaught_crash"]) for row in faults), sum(truth(row["uncaught_crash"]) for row in faults), 0)
    add(checks, "zero_caller_mutation", all(truth(row["caller_state_unchanged"]) for row in faults), sum(not truth(row["caller_state_unchanged"]) for row in faults), 0)
    add(checks, "fault_gate_88_of_88", all(truth(row["pass"]) for row in faults), sum(truth(row["pass"]) for row in faults), 88)

    recomputed = {}
    for axis in ("all", "commission_or_structure", "omission"):
        for arm in ARMS:
            selected = [row for row in faults if row["arm"] == arm and (axis == "all" or row["threat_axis"] == axis)]
            layers = Counter(row["first_rejection_layer"] for row in selected)
            recomputed[(axis, arm)] = {
                "fault_cells": len(selected),
                "parser_rejected": layers["schema_parser"],
                "native_representation_guard_rejected": layers["native_representation_guard"],
                "common_validator_rejected": layers["common_validator"],
                "accepted": layers["accepted"],
                "representation_fail_open": sum(truth(row["representation_fail_open"]) for row in selected),
                "end_to_end_fail_open": sum(truth(row["end_to_end_fail_open"]) for row in selected),
            }
    aggregate_map = {(row["threat_axis"], row["arm"]): row for row in aggregate}
    aggregate_ok = len(aggregate) == 12 and set(aggregate_map) == set(recomputed)
    if aggregate_ok:
        aggregate_ok = all(
            all(int(aggregate_map[key][field]) == value for field, value in values.items())
            for key, values in recomputed.items()
        )
    add(checks, "aggregate_recomputed_from_raw", aggregate_ok, len(aggregate), 12)

    def reps(axis: str) -> dict[str, int]:
        return {arm: recomputed[(axis, arm)]["representation_fail_open"] for arm in sorted(ARMS)}

    expected_all = {"compact_tx": 13, "cope_exact": 0, "cope_generic": 8, "fsr_pc": 18}
    expected_commission = {"compact_tx": 5, "cope_exact": 0, "cope_generic": 0, "fsr_pc": 10}
    expected_omission = {"compact_tx": 8, "cope_exact": 0, "cope_generic": 8, "fsr_pc": 8}
    add(checks, "all_representation_fail_open_counts", reps("all") == expected_all, reps("all"), expected_all)
    add(checks, "commission_representation_fail_open_counts", reps("commission_or_structure") == expected_commission, reps("commission_or_structure"), expected_commission)
    add(checks, "omission_representation_fail_open_counts", reps("omission") == expected_omission, reps("omission"), expected_omission)
    gates_ok = status.get("source_audit_pass") == "True" and status.get("oracle_removal_gate") == "True" and status.get("local_typed_assurance_gate") == "True" and status.get("omission_boundary_triggered") == "True"
    add(checks, "frozen_gates", gates_ok, [status.get("source_audit_pass"), status.get("oracle_removal_gate"), status.get("local_typed_assurance_gate"), status.get("omission_boundary_triggered")], ["True", "True", "True", "True"])
    boundary_ok = status.get("learned_provider_calls") == "0" and status.get("gpu_llm_robot_simulator_used") == "false" and status.get("reserved_states_27_49_read") == "false" and status.get("evidence_class") == "offline_deterministic_fault_injection"
    add(checks, "evidence_resource_boundary", boundary_ok, [status.get("learned_provider_calls"), status.get("gpu_llm_robot_simulator_used"), status.get("reserved_states_27_49_read"), status.get("evidence_class")], ["0", "false", "false", "offline_deterministic_fault_injection"])
    add(checks, "run_decision_pass", status.get("decision") == "PASS", status.get("decision"), "PASS")

    output = ROOT / "02_INDEPENDENT_AUDIT.csv"
    with output.open("x", newline="", encoding="utf-8") as stream:
        writer = csv.DictWriter(stream, fieldnames=["check", "pass", "observed", "expected"], lineterminator="\n")
        writer.writeheader()
        writer.writerows(checks)
    passed = sum(row["pass"] == "True" for row in checks)
    decision = "PASS" if passed == len(checks) else "FAIL"
    result = {
        "audit_program_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "checks_passed": passed,
        "checks_total": len(checks),
        "decision": decision,
        "fault_cells": len(faults),
        "end_to_end_fail_open": sum(truth(row["end_to_end_fail_open"]) for row in faults),
        "learned_provider_calls": 0,
        "reserved_states_read": False,
    }
    (ROOT / "02_INDEPENDENT_AUDIT_STATUS.txt").write_text(json.dumps(result, sort_keys=True, separators=(",", ":")) + "\n", encoding="utf-8")
    print(json.dumps(result, sort_keys=True))
    return 0 if decision == "PASS" else 1


if __name__ == "__main__":
    raise SystemExit(main())
