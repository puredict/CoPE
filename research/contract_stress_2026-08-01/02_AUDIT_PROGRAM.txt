@dataclass(frozen=True)
class Case:
    family: str
    case_id: str
    dimension: str
    expected_accept: bool
    target: str
    op: str
    path: tuple[Any, ...]
    value: Any
    rationale: str
    isolation_required: bool = False


def case(
    family: str,
    case_id: str,
    dimension: str,
    *,
    target: str = "state",
    op: str = "noop",
    path: tuple[Any, ...] = (),
    value: Any = None,
    rationale: str,
    expected_accept: bool = False,
    isolation_required: bool = False,
) -> Case:
    return Case(
        family=family,
        case_id=case_id,
        dimension=dimension,
        expected_accept=expected_accept,
        target=target,
        op=op,
        path=path,
        value=value,
        rationale=rationale,
        isolation_required=isolation_required,
    )

def all_cases() -> list[Case]:

        "cancellation_fullstate_v2", cancellation=True
    ) + legacy_cases()


def resolve(root: Any, path: tuple[Any, ...]) -> Any:
    current = root
    for key in path:
        current = current[key]
    return current


def mutate(root: Any, item: Case, truth: list[str]) -> list[str]:
    if item.op == "noop":
        return truth
    if item.op == "set_truth":
        return list(item.value)
    target = root if item.target == "state" else root["event"]
    parent = resolve(target, item.path[:-1]) if item.path else target
    key = item.path[-1] if item.path else None
    if item.op == "set":
        parent[key] = copy.deepcopy(item.value)
    elif item.op == "pop":
        parent.pop(key)
    elif item.op == "append":
        resolve(target, item.path).append(copy.deepcopy(item.value))
    elif item.op == "remove":
        resolve(target, item.path).pop(int(item.value))
    elif item.op == "duplicate":
        collection = resolve(target, item.path)
        collection.append(copy.deepcopy(collection[int(item.value)]))
    elif item.op == "copy":
        parent[key] = copy.deepcopy(resolve(target, tuple(item.value)))
    else:
        raise ValueError(f"unknown mutation op {item.op!r}")
    return truth

def run_case(item: Case) -> dict[str, Any]:
    if item.family == "replacement_fullstate_v2":
        state, event, truth = replacement_fixture()
        canonical_compiled = compile_controller_prompt(state)
    elif item.family == "cancellation_fullstate_v2":
        state, event, truth = cancellation_fixture()
        canonical_compiled = compile_execution_directive(state)
    else:
        state, event, truth = legacy_fixture()
        canonical_compiled = ""
    bundle = {"state": copy.deepcopy(state), "event": copy.deepcopy(event)}
    truth = mutate(bundle["state"], item, list(truth)) if item.target != "event" else mutate(bundle, item, list(truth))
    state = bundle["state"]
    event = bundle["event"]
    accepted = False
    error_class = ""
    error_message = ""
    isolation_pass: bool | str = ""
    try:
        if item.family == "replacement_fullstate_v2":
            validate_oracle_full_state(
                state,
                event,
                previous_state_version=0,
                physically_true_objects=truth,
            )
            accepted = True
            if item.isolation_required:
                isolation_pass = compile_controller_prompt(state) == canonical_compiled
        elif item.family == "cancellation_fullstate_v2":
            validate_oracle_cancellation_state(
                state,
                event,
                previous_state_version=0,
                physically_true_objects=truth,
            )
            accepted = True
            if item.isolation_required:
                isolation_pass = compile_execution_directive(state) == canonical_compiled
        else:
            FullStateOutput.from_mapping(state)
            accepted = True
    except Exception as exc:
        error_class = type(exc).__name__
        error_message = str(exc)
    expectation_met = accepted == item.expected_accept
    if item.isolation_required:
        expectation_met = expectation_met and isolation_pass is True
    return {
        "family": item.family,
        "case_id": item.case_id,
        "dimension": item.dimension,
        "expected_accept": item.expected_accept,
        "actual_accept": accepted,
        "expectation_met": expectation_met,
        "isolation_required": item.isolation_required,
        "isolation_pass": isolation_pass,
        "error_class": error_class,
        "error_message": error_message,
    }


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--manifest-csv", type=Path)
    group.add_argument("--result-csv", type=Path)
    return parser.parse_args()


def write_rows(path: Path, rows: list[dict[str, Any]]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("x", newline="", encoding="utf-8") as handle:
        writer = csv.DictWriter(handle, fieldnames=list(rows[0]), lineterminator="\n")
        writer.writeheader()
        writer.writerows(rows)


def main() -> int:
    args = parse_args()
    cases = all_cases()
    if args.manifest_csv:
        rows = [
            {
                "family": item.family,
                "case_id": item.case_id,
                "dimension": item.dimension,
                "expected_accept": item.expected_accept,
                "isolation_required": item.isolation_required,
                "target": item.target,
                "operation": item.op,
                "path": json.dumps(item.path, separators=(",", ":")),
                "value": json.dumps(item.value, sort_keys=True, separators=(",", ":")),
                "rationale": item.rationale,
            }
            for item in cases
        ]
        write_rows(args.manifest_csv, rows)
        print(f"manifest_cases={len(rows)}")
        return 0
    rows = [run_case(item) for item in cases]
    write_rows(args.result_csv, rows)
    failures = [row for row in rows if not row["expectation_met"]]
    by_family: dict[str, list[dict[str, Any]]] = {}
    for row in rows:
        by_family.setdefault(str(row["family"]), []).append(row)
    print(f"cases={len(rows)} expectation_failures={len(failures)}")
    for family, family_rows in by_family.items():
        accepted_bad = sum(
            bool(row["actual_accept"]) and not bool(row["expected_accept"])
            for row in family_rows
        )
        print(
            f"{family}: cases={len(family_rows)} "
            f"expectation_failures={sum(not bool(row['expectation_met']) for row in family_rows)} "
            f"malformed_accepted={accepted_bad}"
        )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
