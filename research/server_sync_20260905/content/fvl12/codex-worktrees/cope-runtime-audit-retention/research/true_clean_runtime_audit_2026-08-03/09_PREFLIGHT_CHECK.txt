def run(argv: list[str], *, cwd: Path | None = None) -> subprocess.CompletedProcess[str]:
    return subprocess.run(argv, cwd=cwd, text=True, capture_output=True, check=False)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def tree_content_sha256(root: Path) -> tuple[int, int, str]:
    files: list[tuple[str, Path]] = []
    for path in root.rglob("*"):
        if not path.is_file():
            continue
        rel = path.relative_to(root).as_posix()
        parts = rel.split("/")
        if rel == ".git" or rel.startswith(".git/") or "__pycache__" in parts or rel.endswith((".pyc", ".pyo", ".DS_Store")):
            continue
        files.append((rel, path))
    files.sort()
    aggregate = hashlib.sha256()
    total = 0
    for rel, path in files:
        size = path.stat().st_size
        total += size
        aggregate.update(rel.encode() + b"\0" + str(size).encode() + b"\0" + sha256_file(path).encode() + b"\n")
    return len(files), total, aggregate.hexdigest()
