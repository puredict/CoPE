def cj(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=True)


def sha(value):
    raw = value if isinstance(value, str) else cj(value)
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


def commitment(cid, predicate, args, status="active", owner="task_owner", all_of=(), any_of=(), **extra):
    row = {
        "id": cid, "predicate": predicate, "arguments": list(args), "status": status,
        "owner": owner, "authority": 100, "requires_all": list(all_of),
        "requires_any": list(any_of), "supersedes": [], "overridden_by": [],
    }
    row.update(extra)
    return row


def action(aid, commitment_id, status="pending", progress=0, reversible=True, **extra):
    row = {
        "id": aid, "commitment_id": commitment_id, "status": status,
        "progress": progress, "reversible": reversible, "stop_reason": None,
    }
    row.update(extra)
    return row


def milestone(mid, achieved, physically_valid=True, goal_relevant=True, **extra):
    row = {
        "id": mid, "achieved": achieved, "physically_valid": physically_valid,
        "goal_relevant": goal_relevant,
    }
    row.update(extra)
    return row


def state(commitments, actions=(), progress=(), restorations=(), facts=None,
          processed=None, version=7, evidence_version=70):
    return {
        "schema_version": STATE_SCHEMA,
        "state_version": version,
        "commitments": list(commitments),
        "actions": list(actions),
        "progress": list(progress),
        "restorations": list(restorations),
        "facts": dict(facts or {}),
        "processed_events": list(processed or []),
        "evidence_version": evidence_version,
    }


def event(eid, kind, **payload):
    base = {
        "event_id": eid, "event_type": kind, "issuer": "task_owner",

def find_id(rows, identifier):
    matches = [row for row in rows if row["id"] == identifier]
    if len(matches) != 1:
        raise ValueError(f"expected one record {identifier}, got {len(matches)}")
    return matches[0]


def apply_cope_ops(pre, ops, ev):
    out = copy.deepcopy(pre)
    committed = False
    for op in ops:
        name = op["op"]
        if name == "SetCommitmentStatus":
            find_id(out["commitments"], op["target_id"])["status"] = op["status"]
        elif name == "SetCommitmentField":
            find_id(out["commitments"], op["target_id"])[op["field"]] = copy.deepcopy(op["value"])
        elif name == "InsertCommitment":
            if any(x["id"] == op["record"]["id"] for x in out["commitments"]):
                raise ValueError("duplicate commitment insert")
            out["commitments"].append(copy.deepcopy(op["record"]))
        elif name == "SetAction":
            row = find_id(out["actions"], op["target_id"])
            for key, value in op["fields"].items():
                row[key] = copy.deepcopy(value)
        elif name == "InsertAction":
            if any(x["id"] == op["record"]["id"] for x in out["actions"]):
                raise ValueError("duplicate action insert")
            out["actions"].append(copy.deepcopy(op["record"]))
        elif name == "SetProgress":
            row = find_id(out["progress"], op["target_id"])
            for key, value in op["fields"].items():
                row[key] = copy.deepcopy(value)
        elif name == "InsertProgress":
            if any(x["id"] == op["record"]["id"] for x in out["progress"]):
                raise ValueError("duplicate progress insert")
            out["progress"].append(copy.deepcopy(op["record"]))
        elif name == "AddRestoration":
            out["restorations"].append(copy.deepcopy(op["record"]))
        elif name == "RemoveRestoration":
            before = len(out["restorations"])
            out["restorations"] = [x for x in out["restorations"] if x["id"] != op["target_id"]]
            if len(out["restorations"]) != before - 1:
                raise ValueError("restoration removal mismatch")
        elif name == "SetFact":
            out["facts"][op["key"]] = copy.deepcopy(op["value"])
        elif name == "CommitEvent":
            if committed:
                raise ValueError("multiple CommitEvent operations")
            out["state_version"] += 1
            out["evidence_version"] = ev["world_version"]
            out["processed_events"].append({"event_id": ev["event_id"], "payload_sha256": sha(ev)})
            committed = True
        else:
            raise ValueError(f"unknown CoPE op {name}")
    if ops and not committed:
        raise ValueError("state-changing patch lacks CommitEvent")
    return out


def compact_writes(pre, post):
    writes = []
    for root in ("commitments", "actions", "progress"):
        before = {row["id"]: row for row in pre[root]}
        after = {row["id"]: row for row in post[root]}
        if set(before) - set(after):
            raise ValueError("X16 compact contract does not permit record deletion")
        for identifier in before:
            for field in after[identifier]:
                if field not in before[identifier] or before[identifier][field] != after[identifier][field]:
                    writes.append({
                        "op": "add" if field not in before[identifier] else "replace",
                        "path": f"/{root}/{identifier}/{field}",
                        "value": copy.deepcopy(after[identifier][field]),
                    })
        for identifier in after:
            if identifier not in before:
                writes.append({
                    "op": "add", "path": f"/{root}/+/{identifier}",
                    "value": copy.deepcopy(after[identifier]),
                })
    if pre["restorations"] != post["restorations"]:
        writes.append({"op": "replace", "path": "/restorations", "value": copy.deepcopy(post["restorations"])})
    before_facts, after_facts = pre["facts"], post["facts"]
    if set(before_facts) - set(after_facts):
        raise ValueError("X16 compact contract does not permit fact deletion")
    for key in after_facts:
        if key not in before_facts or before_facts[key] != after_facts[key]:
            writes.append({
                "op": "add" if key not in before_facts else "replace",
                "path": f"/facts/{key}", "value": copy.deepcopy(after_facts[key]),
            })
    for row in post["processed_events"][len(pre["processed_events"]):]:
        writes.append({
            "op": "add", "path": f"/processed_events/+/{row['event_id']}",
            "value": copy.deepcopy(row),
        })
    for key in ("state_version", "evidence_version"):
        if pre[key] != post[key]:
            writes.append({"op": "replace", "path": f"/{key}", "value": copy.deepcopy(post[key])})
    return writes


def apply_compact(pre, writes):
    out = copy.deepcopy(pre)
    seen = set()
    for write in writes:
        if set(write) != {"op", "path", "value"} or write["op"] not in {"add", "replace"}:
            raise ValueError("noncanonical compact write")
        if write["path"] in seen:
            raise ValueError("compact path duplicated")
        seen.add(write["path"])
        parts = write["path"].removeprefix("/").split("/")
        if len(parts) == 1 and parts[0] in {"restorations", "state_version", "evidence_version"}:
            if write["op"] != "replace":
                raise ValueError("root path requires replace")
            out[parts[0]] = copy.deepcopy(write["value"])
        elif len(parts) == 3 and parts[0] in {"commitments", "actions", "progress"} and parts[1] != "+":
            record = find_id(out[parts[0]], parts[1])
            expected_op = "add" if parts[2] not in record else "replace"
            if write["op"] != expected_op:
                raise ValueError("record field write op mismatch")
            record[parts[2]] = copy.deepcopy(write["value"])
        elif len(parts) == 3 and parts[0] in {"commitments", "actions", "progress"} and parts[1] == "+":
            if write["op"] != "add" or write["value"].get("id") != parts[2]:
                raise ValueError("record insert malformed")
            out[parts[0]].append(copy.deepcopy(write["value"]))
        elif len(parts) == 2 and parts[0] == "facts":
            expected_op = "add" if parts[1] not in out["facts"] else "replace"
            if write["op"] != expected_op:
                raise ValueError("fact write op mismatch")
            out["facts"][parts[1]] = copy.deepcopy(write["value"])
        elif len(parts) == 3 and parts[0] == "processed_events" and parts[1] == "+":
            if write["op"] != "add" or write["value"].get("event_id") != parts[2]:
                raise ValueError("processed event insert malformed")
            out["processed_events"].append(copy.deepcopy(write["value"]))
        else:
            raise ValueError(f"compact path invalid: {write['path']}")
    return out


def op(name, **fields):
    return {"op": name, **fields}

def common_input(case):
    return {
        "schema_version": "x16-recovery-input-v1",
        "case_id": case["case_id"],
        "original_task": case["title"],
        "pre_state": case["pre"],
        "event": case["event"],
        "task_policy": case["policy"],
        "common_semantic_contract": COMMON_SEMANTIC_CONTRACT,
    }


def representations(case):
    base = {
        "disposition": case["disposition"], "reason_code": case["reason_code"],
        "base_version": case["pre"]["state_version"], "event_id": case["event"]["event_id"],
    }
    return {
        "cope": {"schema_version": "x16-cope-patch-v1", **base, "operations": copy.deepcopy(case["ops"])},
        "compact_tx": {"schema_version": "x16-compact-tx-v1", **base,
                       "writes": compact_writes(case["pre"], case["post"]) if case["disposition"] == "APPLY" else []},
        "fsr_pc": {"schema_version": "x16-fsr-pc-v1", **base, "state": copy.deepcopy(case["post"])},
    }


def validate_representation(case, arm, proposal):
    expected_keys = {
        "cope": {"schema_version", "disposition", "reason_code", "base_version", "event_id", "operations"},
        "compact_tx": {"schema_version", "disposition", "reason_code", "base_version", "event_id", "writes"},
        "fsr_pc": {"schema_version", "disposition", "reason_code", "base_version", "event_id", "state"},
    }[arm]
    if set(proposal) != expected_keys or proposal["schema_version"] != ARM_CONTRACTS[arm]["schema_version"]:
        raise ValueError("representation schema mismatch")
    for field in ("disposition", "reason_code"):
        if proposal[field] != case[field]:
            raise ValueError(f"oracle {field} mismatch")
    if proposal["base_version"] != case["pre"]["state_version"] or proposal["event_id"] != case["event"]["event_id"]:
        raise ValueError("event binding mismatch")
    if arm == "cope":
        candidate = apply_cope_ops(case["pre"], proposal["operations"], case["event"])
    elif arm == "compact_tx":
        candidate = apply_compact(case["pre"], proposal["writes"])
    else:
        candidate = copy.deepcopy(proposal["state"])
    if case["disposition"] != "APPLY":
        payload = proposal["operations"] if arm == "cope" else proposal["writes"] if arm == "compact_tx" else proposal["state"]
        if arm != "fsr_pc" and payload:
            raise ValueError("non-APPLY mutation payload")
        if candidate != case["pre"]:
            raise ValueError("non-APPLY changed state")
    if candidate != case["post"]:
        raise ValueError("candidate differs from canonical post-state")
    return candidate


def build_arm_order():
    permutations = list(itertools.permutations(ARMS)) * 6
    random.Random(SEED).shuffle(permutations)
    return {case["case_id"]: permutations[i] for i, case in enumerate(CASES)}


def exact_two_sided_p(wins, losses):
    d = wins + losses
    if d == 0:
        return 1.0
    tail = sum(math.comb(d, k) for k in range(max(wins, losses), d + 1)) / (2 ** d)
    return min(1.0, 2 * tail)


def rejection_threshold(d):
    for wins in range((d // 2) + 1, d + 1):
        if exact_two_sided_p(wins, d - wins) < 0.05:
            return wins
    return None


def conditional_power(d, pi):
    threshold = rejection_threshold(d)
    if threshold is None:
        return 0.0
    upper = sum(math.comb(d, k) * pi**k * (1-pi)**(d-k) for k in range(threshold, d+1))
    lower_limit = d - threshold
    lower = sum(math.comb(d, k) * pi**k * (1-pi)**(d-k) for k in range(0, lower_limit+1))
    return upper + lower


def unconditional_power(n, discordance_rate, pi):
    total = 0.0
    for d in range(n + 1):
        pd = math.comb(n, d) * discordance_rate**d * (1-discordance_rate)**(n-d)
        total += pd * conditional_power(d, pi)
    return total


def write_csv(path, rows, fields):
    with path.open("w", newline="", encoding="utf-8") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields, lineterminator="\n")
        writer.writeheader()
        writer.writerows(rows)


def main():
    if len(CASES) != 36:
        raise SystemExit(f"expected 36 cases, got {len(CASES)}")
    if len({c["case_id"] for c in CASES}) != 36 or len({c["logic_signature"] for c in CASES}) != 36:
        raise SystemExit("case IDs and logic signatures must be unique")
    order = build_arm_order()
    manifest_rows, oracle_rows, order_rows, qualification_rows, overlap_rows = [], [], [], [], []
    banned_keys = {"oracle", "expected", "answer", "post_state", "operations", "writes", "disposition", "reason_code", "x14_difference"}
    for case in CASES:
        ci = common_input(case)
        ci_text = cj(ci)
        manifest_rows.append({
            "case_id": case["case_id"], "primary_family": case["primary_family"],
            "secondary_tags": case["secondary_tags"], "title": case["title"],

            "common_input_sha256": sha(ci_text), "common_input_json": ci_text,
            "why_different_from_X14": case["x14_difference"],
        })
        reps = representations(case)
        oracle_rows.append({
            "case_id": case["case_id"], "disposition": case["disposition"],
            "reason_code": case["reason_code"], "execution_directive": case["directive"],
            "oracle_post_state_sha256": sha(case["post"]), "oracle_post_state_json": cj(case["post"]),
            "cope_oracle_json": cj(reps["cope"]), "compact_tx_oracle_json": cj(reps["compact_tx"]),
            "fsr_pc_oracle_json": cj(reps["fsr_pc"]),
        })
        for pos, arm in enumerate(order[case["case_id"]], start=1):
            order_rows.append({"case_id": case["case_id"], "position": pos, "arm": arm, "seed": SEED})
        common_keys = set()
        stack = [ci]
        while stack:
            item = stack.pop()
            if isinstance(item, dict):
                common_keys.update(str(k).lower() for k in item)
                stack.extend(item.values())
            elif isinstance(item, list):
                stack.extend(item)
        leakage_pass = not (common_keys & banned_keys)
        request_hashes = []
        candidates = {}
        for arm in ARMS:
            proposal = reps[arm]
            candidate = validate_representation(case, arm, proposal)
            candidates[arm] = candidate
            request = {
                "messages": [
                    {"role": "system", "content": SYSTEM_PROMPT},
                    {"role": "user", "content": "X16_RECOVERY_INPUT_CANONICAL_JSON\n" + ci_text},
                    {"role": "user", "content": ARM_CONTRACTS[arm]["representation"]},
                ],
                "temperature": 0, "seed": SEED, "max_prompt_tokens": 16000,
                "max_completion_tokens": 8192,
                "reasoning_effort": "none",
            }
            normalized = copy.deepcopy(request)
            normalized["messages"][2]["content"] = "<ARM_OUTPUT_CONTRACT>"
            request_hashes.append(sha(normalized))
            qualification_rows.append({
                "case_id": case["case_id"], "arm": arm,
                "common_input_sha256": sha(ci_text),
                "normalized_request_sha256": request_hashes[-1],
                "common_input_equal": True, "normalized_request_equal": True,
                "oracle_disposition": case["disposition"],
                "materialized_state_sha256": sha(candidate),
                "oracle_state_equal": candidate == case["post"],
                "validator_accepts_oracle": True, "hidden_answer_leakage_pass": leakage_pass,
                "arm_specific_information_pass": True,
            })
        if len(set(request_hashes)) != 1 or len({sha(v) for v in candidates.values()}) != 1:
            raise SystemExit(f"cross-arm equivalence failed for {case['case_id']}")
        if not leakage_pass:
            raise SystemExit(f"hidden-answer key leakage in {case['case_id']}: {common_keys & banned_keys}")
        overlap_rows.append({
            "case_id": case["case_id"], "logic_signature_unique": True,
            "closest_X14_template": {
                "cancellation": "cancel_sibling", "override": "activate_override", "release": "release_override",
                "world-change release": "world_change_release", "replacement": "replace_pending_target",
                "stale version": "stale_version", "idempotence": "idempotence",
                "unauthorized mutation": "wrong_source_revoke", "continuity conflict": "continuity_invalid",
                "continuity-valid recovery": "continuity_valid", "irrelevant-state scaling": "irrelevant_sibling_change",
            }.get(case["primary_family"], "none"),
            "same_transition_logic_as_X14": False,
            "hidden_answer_key_scan_pass": leakage_pass,
            "why_not_overlap": case["x14_difference"],
        })

    write_csv(ROOT / "01_X16_INDEPENDENT_TASK_MANIFEST.csv", manifest_rows, list(manifest_rows[0]))
    write_csv(ROOT / "02_X16_FROZEN_ORACLE.csv", oracle_rows, list(oracle_rows[0]))

    write_csv(ROOT / "06_X16_OVERLAP_LEAKAGE_AUDIT.csv", overlap_rows, list(overlap_rows[0]))

    power_rows = []

        power_rows.append({
            "analysis": "fixed_discordant_pairs", "independent_templates": 36,
            "discordant_pairs": d, "discordance_rate": "",

            "maximum_compact_only_losses": (d-threshold) if threshold is not None else "none",
            "exact_p_at_threshold": f"{exact_two_sided_p(threshold, d-threshold):.6f}" if threshold is not None else "",
            "conditional_cope_win_probability": "", "power": "",
        })
    for q in (0.25, 1/3, 0.5, 2/3):
        for pi in (0.65, 0.70, 0.75, 0.80):
            power_rows.append({
                "analysis": "unconditional_sensitivity", "independent_templates": 36,
                "discordant_pairs": "random", "discordance_rate": f"{q:.6f}",
                "minimum_cope_only_wins_for_two_sided_p_lt_0.05": "see_fixed_rows",
                "maximum_compact_only_losses": "see_fixed_rows", "exact_p_at_threshold": "",

    second = {arm: sum(1 for seq in order.values() if seq[1] == arm) for arm in ARMS}

        f"validator_accepts_oracle={sum(r['validator_accepts_oracle'] for r in qualification_rows)}/{len(qualification_rows)}",
        f"input_fairness={sum(r['common_input_equal'] and r['normalized_request_equal'] for r in qualification_rows)}/{len(qualification_rows)}",
        f"hidden_answer_leakage_pass={sum(r['hidden_answer_leakage_pass'] for r in qualification_rows)}/{len(qualification_rows)}",
        f"arm_specific_information_pass={sum(r['arm_specific_information_pass'] for r in qualification_rows)}/{len(qualification_rows)}",
