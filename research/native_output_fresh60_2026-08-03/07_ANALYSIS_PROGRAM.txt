#!/usr/bin/env python3
from __future__ import annotations

import argparse
import csv
import json
import math
from collections import Counter, defaultdict
from pathlib import Path


ARMS = ("cope", "compact_tx", "fsr_pc")
SUBSTANTIVE = {
    "replacement", "continuity_conflict", "cancellation", "override",
    "release", "release_world_change", "continuity_valid",
}


def yes(value: str) -> bool:
    return str(value).lower() == "true"


def exact_sign_p(wins: int, losses: int) -> float:
    n = wins + losses
    if not n:
        return 1.0
    tail = sum(math.comb(n, k) for k in range(min(wins, losses) + 1)) / (2 ** n)
    return min(1.0, 2 * tail)


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--manifest", type=Path, required=True)
    parser.add_argument("--run-dir", type=Path, required=True)
    parser.add_argument("--output-dir", type=Path, required=True)
    args = parser.parse_args()
    args.output_dir.mkdir(parents=True, exist_ok=False)
    manifest = list(csv.DictReader(args.manifest.open(newline="", encoding="utf-8")))
    manifest_by_case = {row["case_id"]: row for row in manifest}
    rows = list(csv.DictReader((args.run_dir / "04_PER_SAMPLE_RESULTS.csv").open(newline="", encoding="utf-8")))
    fair = list(csv.DictReader((args.run_dir / "03_FAIRNESS_INPUT_HASH_AUDIT.csv").open(newline="", encoding="utf-8")))
    status = json.loads((args.run_dir / "07_RUN_STATUS.txt").read_text(encoding="utf-8"))
    by_case = defaultdict(dict)
    for row in rows:
        by_case[row["case_id"]][row["arm"]] = row
    cluster_rows = []
    for base_case_id in dict.fromkeys(row["base_case_id"] for row in manifest):
        case_ids = [row["case_id"] for row in manifest if row["base_case_id"] == base_case_id]
        family = next(row["family"] for row in manifest if row["base_case_id"] == base_case_id)
        counts = {
            arm: sum(yes(by_case[case_id][arm]["semantic_correct"]) for case_id in case_ids)
            for arm in ARMS
        }
        difference = counts["cope"] - counts["compact_tx"]
        cluster_rows.append({
            "base_case_id": base_case_id,
            "family": family,
            "variants": len(case_ids),
            "cope_correct": counts["cope"],
            "compact_tx_correct": counts["compact_tx"],
            "fsr_pc_correct": counts["fsr_pc"],
            "cope_minus_compact": difference,
            "cluster_outcome": "cope_win" if difference > 0 else "compact_win" if difference < 0 else "tie",
        })
    with (args.output_dir / "08_CLUSTER_RESULTS.csv").open("x", newline="", encoding="utf-8") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(cluster_rows[0]), lineterminator="\n")
        writer.writeheader()
        writer.writerows(cluster_rows)
    correct = {
        arm: sum(row["arm"] == arm and yes(row["semantic_correct"]) for row in rows)
        for arm in ARMS
    }
    safety = {
        arm: sum(
            row["arm"] == arm and any(
                yes(row[field]) for field in ("unauthorized_or_stale_edit", "progress_corruption", "continuity_error")
            )
            for row in rows
        )
        for arm in ARMS
    }
    tokens = {arm: sum(int(row["completion_tokens"]) for row in rows if row["arm"] == arm) for arm in ARMS}
    latency = {arm: sum(float(row["latency_seconds"]) for row in rows if row["arm"] == arm) for arm in ARMS}
    proposal_bytes = {arm: sum(int(row["proposal_bytes"]) for row in rows if row["arm"] == arm) for arm in ARMS}
    cluster_counts = Counter(row["cluster_outcome"] for row in cluster_rows)
    cell_cope_only = 0
    cell_compact_only = 0
    substantive_cope_only = 0
    for case_id, triplet in by_case.items():
        cope_ok = yes(triplet["cope"]["semantic_correct"])
        compact_ok = yes(triplet["compact_tx"]["semantic_correct"])
        cell_cope_only += int(cope_ok and not compact_ok)
        cell_compact_only += int(compact_ok and not cope_ok)
        substantive_cope_only += int(
            cope_ok and not compact_ok and manifest_by_case[case_id]["family"] in SUBSTANTIVE
        )
    position_rows = []
    for arm in ARMS:
        for position in range(3):
            selected = [row for row in rows if row["arm"] == arm and int(row["call_order_position"]) == position]
            position_rows.append({
                "arm": arm, "position": position, "assigned": len(selected),
                "correct": sum(yes(row["semantic_correct"]) for row in selected),
            })
    with (args.output_dir / "09_POSITION_RESULTS.csv").open("x", newline="", encoding="utf-8") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(position_rows[0]), lineterminator="\n")
        writer.writeheader()
        writer.writerows(position_rows)
    all_provider_ok = len(rows) == 180 and all(row["provider_status"] == "ok" for row in rows)
    all_fair = len(fair) == 60 and all(yes(row["fairness_pass"]) for row in fair)
    decision = bool(
        status.get("expanded") is True and all_provider_ok and all_fair
        and correct["cope"] > correct["compact_tx"]
        and cluster_counts["cope_win"] > cluster_counts["compact_win"]
        and safety["cope"] <= safety["compact_tx"]
        and substantive_cope_only > 0
    )
    summary = [{
        "assigned_triplets": len(by_case),
        "provider_ok_cells": sum(row["provider_status"] == "ok" for row in rows),
        "fair_triplets": sum(yes(row["fairness_pass"]) for row in fair),
        "cope_correct": correct["cope"],
        "compact_tx_correct": correct["compact_tx"],
        "fsr_pc_correct": correct["fsr_pc"],
        "cope_only_cells": cell_cope_only,
        "compact_only_cells": cell_compact_only,
        "cell_exact_sign_p_exploratory": f"{exact_sign_p(cell_cope_only, cell_compact_only):.8f}",
        "cope_win_clusters": cluster_counts["cope_win"],
        "compact_win_clusters": cluster_counts["compact_win"],
        "tie_clusters": cluster_counts["tie"],
        "cluster_exact_sign_p": f"{exact_sign_p(cluster_counts['cope_win'], cluster_counts['compact_win']):.8f}",
        "cope_safety_failures": safety["cope"],
        "compact_safety_failures": safety["compact_tx"],
        "fsr_safety_failures": safety["fsr_pc"],
        "substantive_cope_only_cells": substantive_cope_only,
        "cope_completion_tokens": tokens["cope"],
        "compact_completion_tokens": tokens["compact_tx"],
        "fsr_completion_tokens": tokens["fsr_pc"],
        "cope_proposal_bytes": proposal_bytes["cope"],
        "compact_proposal_bytes": proposal_bytes["compact_tx"],
        "fsr_proposal_bytes": proposal_bytes["fsr_pc"],
        "cope_latency_seconds": f"{latency['cope']:.6f}",
        "compact_latency_seconds": f"{latency['compact_tx']:.6f}",
        "fsr_latency_seconds": f"{latency['fsr_pc']:.6f}",
        "decision": "GO_TRUE_CLEAN_PREPARATION" if decision else "NO_GO",
    }]
    with (args.output_dir / "10_SUMMARY.csv").open("x", newline="", encoding="utf-8") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(summary[0]), lineterminator="\n")
        writer.writeheader()
        writer.writerows(summary)
    print(canonical := json.dumps(summary[0], sort_keys=True, separators=(",", ":")))
    (args.output_dir / "11_STATUS.txt").write_text(canonical + "\n", encoding="utf-8")
    return 0 if decision else 2


if __name__ == "__main__":
    raise SystemExit(main())
