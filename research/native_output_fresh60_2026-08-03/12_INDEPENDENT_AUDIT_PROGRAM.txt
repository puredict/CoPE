#!/usr/bin/env python3
from __future__ import annotations

import csv
import json
import sys
from collections import Counter
from pathlib import Path


root = Path(sys.argv[1])
run = root / "run_openrouter_x14_v1"
analysis = root / "analysis_v1"
out = Path(sys.argv[2])
rows = list(csv.DictReader((run / "04_PER_SAMPLE_RESULTS.csv").open(newline="", encoding="utf-8")))
fair = list(csv.DictReader((run / "03_FAIRNESS_INPUT_HASH_AUDIT.csv").open(newline="", encoding="utf-8")))
status = json.loads((run / "07_RUN_STATUS.txt").read_text(encoding="utf-8"))
summary = next(csv.DictReader((analysis / "10_SUMMARY.csv").open(newline="", encoding="utf-8")))
clusters = list(csv.DictReader((analysis / "08_CLUSTER_RESULTS.csv").open(newline="", encoding="utf-8")))
positions = list(csv.DictReader((analysis / "09_POSITION_RESULTS.csv").open(newline="", encoding="utf-8")))
checks = [
    ("rows_180", len(rows) == 180, str(len(rows))),
    ("triplets_60", status["assigned_triplets"] == 60, str(status["assigned_triplets"])),
    ("fresh_manifest_status", status["manifest_kind"] == "fresh60", status["manifest_kind"]),
    ("reasoning_none", status["reasoning_effort"] == "none" and all(r["reasoning_effort"] == "none" for r in rows), ""),
    ("smoke_pass_expanded", status["smoke_gate_pass"] is True and status["expanded"] is True, ""),
    ("provider_ok_180", all(r["provider_status"] == "ok" for r in rows), ""),
    ("fair_60", len(fair) == 60 and all(r["fairness_pass"] == "True" for r in fair), str(len(fair))),
    ("response_hashes_180_unique", len({r["response_sha256"] for r in rows}) == 180, ""),
    ("cope_28", summary["cope_correct"] == "28", summary["cope_correct"]),
    ("compact_18", summary["compact_tx_correct"] == "18", summary["compact_tx_correct"]),
    ("fsr_5", summary["fsr_pc_correct"] == "5", summary["fsr_pc_correct"]),
    ("paired_cells_22_12", summary["cope_only_cells"] == "22" and summary["compact_only_cells"] == "12", ""),
    ("clusters_4_2_6", summary["cope_win_clusters"] == "4" and summary["compact_win_clusters"] == "2" and summary["tie_clusters"] == "6", ""),
    ("no_cope_compact_safety", summary["cope_safety_failures"] == "0" and summary["compact_safety_failures"] == "0", ""),
    ("substantive_wins_21", summary["substantive_cope_only_cells"] == "21", summary["substantive_cope_only_cells"]),
    ("positions_20_each", len(positions) == 9 and all(r["assigned"] == "20" for r in positions), ""),
    ("cluster_rows_12", len(clusters) == 12, str(len(clusters))),
    ("decision_go_preparation", summary["decision"] == "GO_TRUE_CLEAN_PREPARATION", summary["decision"]),
    ("credential_not_logged_flag", status["credential_logged"] is False, str(status["credential_logged"])),
]
with out.open("x", newline="", encoding="utf-8") as stream:
    writer = csv.writer(stream, lineterminator="\n")
    writer.writerow(["check", "pass", "detail"])
    writer.writerows(checks)
print(f"checks_pass={sum(item[1] for item in checks)}/{len(checks)}")
raise SystemExit(0 if all(item[1] for item in checks) else 2)
